# macOS-style desktop.

# WhiteSur's install.sh needs two things a container does not provide, and its
# `set -e` turns both into a bare "A failure occurred in package()":
#   TERM       - the progress spinner shells out to tput
#   SUDO_USER  - the user is resolved via logname, then passed to getent

whitesur-icon-theme        TERM=xterm SUDO_USER=builder
whitesur-cursor-theme-git  TERM=xterm SUDO_USER=builder
